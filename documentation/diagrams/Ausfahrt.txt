@startuml
actor Gelegenheitsnutzer
actor Dauermieter
participant "Parkhaus-System" as System
participant "Schranke" as Gate

Gelegenheitsnutzer -> System : Ticket scannen
System -> System : Ticket prüfen

alt gültig
  System -> Gate : Öffnen
  Gate -> Gelegenheitsnutzer : Ausfahrt
else ungültig
  System -> Gelegenheitsnutzer : Ausfahrt verweigert
end

Dauermieter -> System : Code eingeben
System -> System : Code prüfen

alt gültig
  System -> Gate : Öffnen
  Gate -> Dauermieter : Ausfahrt
else ungültig
  System -> Dauermieter : Ausfahrt verweigert
end

@enduml
