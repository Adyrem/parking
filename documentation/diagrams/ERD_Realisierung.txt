@startuml

skinparam defaultFontSize 12
skinparam padding 5

entity ParkingGarage {
  +id : int <<PK>>
  --
  name : string
}

entity Level {
  +id : int <<PK>>
  --
  number : int
  garage_id : int <<FK>>
}

entity ParkingSpot {
  +id : int <<PK>>
  --
  number : int
  is_occupied : boolean
  level_id : int <<FK>>
}

entity Ticket {
  +id : uuid <<PK>>
  --
  entry_time : datetime
  exit_time : datetime
  paid : boolean
  spot_id : int <<FK>>
  pricing_id : int <<FK>>
  permanent_user_id : int <<FK, nullable>>
}

entity Pricing {
  +id : int <<PK>>
  --
  type : string
  garage_id : int <<FK>>
}

entity TimeBasedPricing {
  +pricing_id : int <<PK, FK>>
  --
  time_slots : jsonb
  weekend_time_slots : jsonb
  holiday_time_slots : jsonb
  holidays : jsonb
  rate_per_hour : float
  daily_rate : float
}

entity DailyRatePricing {
  +pricing_id : int <<PK, FK>>
  --
  daily_rate : float
}

entity MonthlyRentPricing {
  +pricing_id : int <<PK, FK>>
  --
  monthly_rent : float
}

entity Payment {
  +id : int <<PK>>
  --
  amount : decimal
  timestamp : datetime
  ticket_id : uuid <<FK>>
}

entity User {
  +id : int <<PK>>
  --
  type : string
}

entity PermanentUser {
  +id : int <<PK, FK>>
  --
  name : string
  access_code : string
  is_blocked : boolean
  rent_paid_until : date
  last_rent_payment_at : datetime
  spot_id : int <<FK>>
}

entity Settings {
  +id : int <<PK>>
  --
  key : string
  value : string
}

ParkingGarage ||--o{ Level
ParkingGarage ||--o{ Pricing
Level ||--o{ ParkingSpot
ParkingSpot ||--o{ Ticket
Pricing ||--o{ Ticket

Pricing ||--o| TimeBasedPricing
Pricing ||--o| DailyRatePricing
Pricing ||--o| MonthlyRentPricing

User ||--|| PermanentUser
PermanentUser |o--o{ Ticket
PermanentUser ||--o| ParkingSpot

Ticket ||--o{ Payment

@enduml
